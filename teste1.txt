*{ Teste 1 - cenario de controle de estoque, semanticamente valido.
   Exercita toda a linguagem: aritmetica inteira e real, memorias
   tipadas, historico (N RES), logica booleana e as estruturas de
   controle (WHILE, IF, IFELSE). Tres tipos: int, real, bool. }*
(START)

*{ aritmetica inteira }*
(48 6 +)                 *{ 54 : int }*
(48 6 -)                 *{ 42 }*
(7 9 *)                  *{ 63 }*
(48 7 /)                 *{ 6  : divisao inteira }*
(48 7 %)                 *{ 6  : resto inteiro }*
(3 4 ^)                  *{ 81 : potencia }*

*{ aritmetica real (o operador | exige reais) }*
(18.0 4.0 |)             *{ 4.5 : divisao real }*
((2.0 3.5 +) (1.5 2.0 *) |)   *{ aninhamento real }*

*{ memoria tipada e historico }*
(40 PECAS)               *{ define PECAS : int }*
((PECAS) 6 *)            *{ usa PECAS numa expressao }*
(2 RES)                  *{ resultado de 2 instrucoes atras }*

*{ booleanos: relacional, AND, OR, NOT }*
((PECAS) 30 >=)          *{ relacional -> bool }*
(((PECAS) 0 >) ((PECAS) 100 <=) AND)   *{ conjuncao aninhada }*
((FALSE TRUE OR) NOT)    *{ disjuncao negada }*
(((PECAS) 100 >) ALERTA) *{ ALERTA : bool }*

*{ controle: laco crescente, decisao simples e decisao dupla }*
(((PECAS) 80 <) (((PECAS) 5 +) PECAS) WHILE)   *{ incrementa de 5 em 5 }*
(((PECAS) 50 ==) (999 FALTA) IF)               *{ IF de corpo int }*
((ALERTA) (1 STATUS) (0 STATUS) IFELSE)        *{ ramos int coincidentes }*
(END)
