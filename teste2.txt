*{ Teste 2 - leitura de sensores, semanticamente valido. Foco em ponto
   flutuante (real), com um bloco inteiro a parte para os operadores que
   exigem inteiros (/ % ^). Memorias real e int, historico, logica e
   as tres estruturas de controle. }*
(START)

*{ ponto flutuante }*
(12.5 7.5 +)                  *{ 20.0 : real }*
(9.0 2.5 -)                   *{ 6.5 }*
(1.5 4.0 *)                   *{ 6.0 }*
(20.0 8.0 |)                  *{ 2.5 : divisao real }*
((6.0 2.0 +) (10.0 5.0 -) |)  *{ aninhamento real }*

*{ bloco inteiro: / % ^ operam apenas sobre inteiros }*
(29 4 /)                      *{ 7 }*
(29 4 %)                      *{ 1 }*
(2 6 ^)                       *{ 64 }*
((29 4 /) (29 4 %) +)         *{ aninhamento inteiro }*

*{ memoria real e historico }*
(36.6 TEMP)                   *{ define TEMP : real }*
((TEMP) 0.4 +)                *{ usa TEMP }*
(1 RES)                       *{ instrucao anterior }*

*{ logica booleana e variavel booleana }*
(((TEMP) 37.0 <) (TRUE FALSE OR) AND)   *{ conjuncao -> bool }*
(((TEMP) 35.0 >=) ESTAVEL)              *{ ESTAVEL : bool }*

*{ controle: laco real decrescente, decisao dupla e simples }*
(((TEMP) 30.0 >) (((TEMP) 1.5 -) TEMP) WHILE)      *{ decrementa 1.5 }*
(((TEMP) 30.0 <=) (1 PRONTO) (0 PRONTO) IFELSE)    *{ ramos int }*
(((ESTAVEL) NOT) (5 ALARME) IF)                    *{ cond bool negada }*
(END)
